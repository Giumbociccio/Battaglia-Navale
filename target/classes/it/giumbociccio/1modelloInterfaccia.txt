public boolean occupato(int[] cella, Campo c){
  boolean occupato;
  
  for (int i = cella[0] - 1; i < cella[0] + 2 && !occupato; i++){
    for (int j = cella[1] - 1; j < cella[1] + 2 && !occupato; j++){
      if(c.getMatrice()[i][j] == 0){
        occupato = false;
      } else {
        occupato = true;
      }
    }
  }
  return occupato;
}
